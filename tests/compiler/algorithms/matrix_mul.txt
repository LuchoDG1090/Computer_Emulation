/* Multiplicación de matrices (3x2) * (2x4)
   Uso de arreglos planos (row-major) y funciones/while del lenguaje de alto nivel.
*/

func mat_mul(int A[], int B[], int C[], int rowsA, int colsA, int colsB):
    int i = 0;
    int j = 0;
    int k = 0;
    int sum = 0;
    int idxA = 0;
    int idxB = 0;
    int idxC = 0;

    i = 0;
    while (i < rowsA):
        j = 0;
        while (j < colsB):
            sum = 0;
            k = 0;
            while (k < colsA):
                idxA = i * colsA + k;        # posición A[i][k]
                idxB = k * colsB + j;        # posición B[k][j]
                sum = sum + A[idxA] * B[idxB];
                k = k + 1;
            endwhile
            idxC = i * colsB + j;            # posición C[i][j]
            C[idxC] = sum;
            j = j + 1;
        endwhile
        i = i + 1;
    endwhile
endfunc
