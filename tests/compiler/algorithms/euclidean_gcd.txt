/* Algoritmo de Euclides:
    Maximo comun divisor entre dos numeros enteros > 0: a y b
*/

# Algoritmo de Euclides
func maxcd(int c, int d):
    while (c != d):
        if (c > d):
            c = c - d;
        else:
            d = d - c;
        endif
    endwhile
    return c;
endfunc

# Leer primer numero
int a;
output "Ingrese el primer numero:";
input a;

# Leer segundo numero
int b;
output "Ingrese el segundo numero:";
input b;

a = maxcd(a, b);

# Imprimir resultado (el MCD esta en 'a')
output "El resultado es:";
output a;
