/* Ordenamiento Burbuja (Bubble Ordering)
   Código en el lenguaje de alto nivel usado en el proyecto.
   Implementa bubble sort con arreglos, swaps y salida de resultados.
*/

func bubble_sort(int arr[], int n):
    int i = 0;
    int j = 0;
    int swapped = 1;
    int tmp = 0;

    while (swapped != 0):
        swapped = 0;
        i = 1;
        while (i < n):
            if (arr[i-1] > arr[i]):
                tmp = arr[i-1];
                arr[i-1] = arr[i];
                arr[i] = tmp;
                swapped = 1;
            endif
            i = i + 1;
        endwhile
        n = n - 1;
    endwhile
endfunc

# Ejemplo de uso
# Declaración e inicialización de un arreglo de 6 elementos
int n = 6;
int arr[6];
arr[0] = 64;
arr[1] = 34;
arr[2] = 25;
arr[3] = 12;
arr[4] = 22;
arr[5] = 11;

# Ejecutar el ordenamiento
bubble_sort(arr, n);
