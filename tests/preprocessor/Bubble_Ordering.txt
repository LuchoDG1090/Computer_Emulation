/* Ordenamiento Burbuja (Bubble Ordering)
   Codigo en el lenguaje de alto nivel usado en el proyecto.
   Implementa bubble sort con arreglos, swaps y salida de resultados.
*/

#include <bubble_sort>

# Ejemplo de uso
output "Ingrese la cantidad de elementos:";
int n = 0;
input n;

# Declaramos un arreglo con capacidad dinamica
int arr[n];

output "Ingrese los elementos separados por espacio:";
int k = 0;
int val = 0;
while (k < n):
    input val;
    arr[k] = val;
    k = k + 1;
endwhile

# Ejecutar el ordenamiento
bubble_sort(arr, n);

# Imprimir arreglo ordenado
output "Arreglo ordenado:";
k = 0;
while (k < n):
    print(arr[k]);
    output " ";
    k = k + 1;
endwhile
output "\n";
