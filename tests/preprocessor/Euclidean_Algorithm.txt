/* Algoritmo de Euclides:
    Máximo comun divisor entre dos numeros enteros > 0: a y b
*/

#include <Euclidean_Algorithm_maxcd>

# Leer primer número
int a;
output "Ingrese el primer numero: ";
input a;

# Leer segundo número
int b;
output "Ingrese el segundo numero: ";
input b;

a = maxcd(a, b);

# Imprimir resultado (el MCD está en 'a')
output "El resultado es: ";
output a;
