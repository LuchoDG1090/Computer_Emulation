# ----------------------------------------
# 1. ESTRUCTURA DEL PROGRAMA
# ----------------------------------------

program = { declaration | func_decl | adt_decl } ;


# ----------------------------------------
# 2. TIPOS DE DATOS
# ----------------------------------------

type = INT_TYPE | FLOAT_TYPE | STRING_TYPE | BOOL_TYPE | CHAR_TYPE | VOID_TYPE ;


# ----------------------------------------
# 3. DECLARACIONES
# ----------------------------------------

# Variables simples y arrays
declaration = type ID [ array_suffix ] [ ASSIGN exp ] ;

# Arrays multidimensionales: int arr[10], float matrix[3][4]
array_suffix = LBRACKET exp RBRACKET { LBRACKET exp RBRACKET } ;


# ----------------------------------------
# 4. FUNCIONES
# ----------------------------------------

# func nombre(params): ... endfunc
func_decl = FUNC ID LPAREN [ param_list ] RPAREN COLON block ENDFUNC ;

param_list = param { COMMA param } ;
param = type ID [ array_suffix ] ;


# ----------------------------------------
# 5. ADTs (TIPOS ABSTRACTOS DE DATOS)
# ----------------------------------------

# adt Nombre(params): ... endadt
adt_decl = ADT ID LPAREN [ param_list ] RPAREN COLON adt_body ENDADT ;

adt_body = { visibility_block | attribute_decl | func_decl } ;

# Bloques de visibilidad: public: / private:
visibility_block = ( PRIVATE | PUBLIC ) COLON { attribute_decl | func_decl } ;

# Atributos del ADT
attribute_decl = type ID [ array_suffix ] ;


# ----------------------------------------
# 6. BLOQUES Y SENTENCIAS
# ----------------------------------------

block = { statement } ;

statement =
      declaration          # int x = 5
    | assign_stmt          # x = 10
    | if_stmt              # if (...): ... endif
    | while_stmt           # while (...): ... endwhile
    | for_stmt             # for (int i = 0; i < 10; i++): ... endfor
    | func_call            # foo(a, b)
    | return_stmt          # return exp
    | BREAK                # break
    | CONTINUE             # continue
    | output_stmt          # output exp
    | input_stmt ;         # input var


# ----------------------------------------
# 7. ASIGNACIÓN
# ----------------------------------------

assign_stmt = lvalue ASSIGN exp ;

# lvalue soporta: x, arr[i], matrix[i][j], obj.attr, obj.arr[i]
lvalue = ID { LBRACKET exp RBRACKET | DOT ID } ;


# ----------------------------------------
# 8. ENTRADA / SALIDA
# ----------------------------------------

output_stmt = OUTPUT exp ;
input_stmt = INPUT lvalue ;


# ----------------------------------------
# 9. ESTRUCTURAS DE CONTROL
# ----------------------------------------

# IF
if_stmt = IF LPAREN exp RPAREN COLON block [ ELSE COLON block ] ENDIF ;

# WHILE
while_stmt = WHILE LPAREN exp RPAREN COLON block ENDWHILE ;

# FOR con rangos: for(int i = 1; i <= x; i++):
for_stmt = FOR LPAREN declaration SEMI exp SEMI exp RPAREN COLON block ENDFOR ;

# RETURN
return_stmt = RETURN [ exp ] ;


# ----------------------------------------
# 10. LLAMADAS A FUNCIONES
# ----------------------------------------

func_call = ID LPAREN [ arg_list ] RPAREN ;
arg_list = exp { COMMA exp } ;


# ----------------------------------------
# 11. EXPRESIONES
# ----------------------------------------

# Precedencia de operadores

exp = logical_or_exp ;

# 1. OR lógico: ||
logical_or_exp = logical_and_exp { OR logical_and_exp } ;

# 2. AND lógico: &&
logical_and_exp = bitwise_or_exp { AND bitwise_or_exp } ;

# 3. OR bitwise: |
bitwise_or_exp = bitwise_xor_exp { BIT_OR bitwise_xor_exp } ;

# 4. XOR bitwise: ^
bitwise_xor_exp = bitwise_and_exp { BIT_XOR bitwise_and_exp } ;

# 5. AND bitwise: &
bitwise_and_exp = equality_exp { BIT_AND equality_exp } ;

# 6. Igualdad: ==, !=
equality_exp = relational_exp { ( EQ | NEQ ) relational_exp } ;

# 7. Relacionales: <, <=, >, >=
relational_exp = shift_exp { ( LT | LE | GT | GE ) shift_exp } ;

# 8. Shifts: <<, >>
shift_exp = additive_exp { ( SHIFT_LEFT | SHIFT_RIGHT ) additive_exp } ;

# 9. Suma/Resta: +, -
additive_exp = multiplicative_exp { ( PLUS | MINUS ) multiplicative_exp } ;

# 10. Multiplicación/División: *, /, %
multiplicative_exp = power_exp { ( TIMES | DIVIDE | MOD ) power_exp } ;

# 11. Potencia: **
power_exp = unary_exp { POWER unary_exp } ;

# 12. Operadores unarios: !, -, +, ~, ++, --
unary_exp = [ NOT | MINUS | PLUS | BIT_NOT | INCREMENT | DECREMENT ] postfix_exp ;

# 13. Expresiones postfijas (para ++ y -- sufijos)
postfix_exp = primary_exp { INCREMENT | DECREMENT } ;

# 14. Expresiones primarias
primary_exp =
      INTEGER              # 42
    | FLOAT                # 3.14, 1.5e10
    | STRING               # "hello", 'world'
    | TRUE                 # true
    | FALSE                # false
    | lvalue               # x, arr[i], obj.attr
    | func_call            # foo(a, b)
    | adt_constructor      # Person("Alice", 30)
    | LPAREN exp RPAREN ;  # (exp)


# ----------------------------------------
# 15. CONSTRUCTORES DE ADT
# ----------------------------------------

# Instanciación: Person p = Person("Alice", 30)
adt_constructor = ID LPAREN [ arg_list ] RPAREN ;
