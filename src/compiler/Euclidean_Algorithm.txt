/* Algoritmo de Euclides:
    Máximo comun divisor entre dos numeros enteros > 0: a y b
    Versión iterativa
    Argumentos: int a0, b0
*/

func maxcd(int a0,int b0):
    int a=a0 # Variable entera a
    int b=a0 # Variable entera b
    _wile (a!=b):
        if(a>b): a=a-b
        else: a=a-b
    endwhile
    return a  
endfunc
